cmake_minimum_required(VERSION 3.16)
project(pocket_sqlite C)
file(GLOB SQLITE_AMALG "${CMAKE_CURRENT_SOURCE_DIR}/sqlite-amalgamation-*/sqlite3.c")
list(LENGTH SQLITE_AMALG SQLITE_AMALG_COUNT)
if(NOT SQLITE_AMALG_COUNT EQUAL 1)
    message(FATAL_ERROR "Stage exactly one pinned SQLite amalgamation")
endif()
get_filename_component(SQLITE_INCLUDE "${SQLITE_AMALG}" DIRECTORY)
add_library(sqlite3 STATIC ${SQLITE_AMALG})
set_target_properties(sqlite3 PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(sqlite3 PUBLIC "${SQLITE_INCLUDE}")
target_compile_definitions(sqlite3 PRIVATE
    SQLITE_ENABLE_FTS5 SQLITE_ENABLE_RTREE SQLITE_ENABLE_COLUMN_METADATA
    SQLITE_ENABLE_JSON1 SQLITE_DEFAULT_MEMSTATUS=0 SQLITE_LIKE_DOESNT_MATCH_BLOBS
    SQLITE_OMIT_DEPRECATED SQLITE_USE_URI=1 SQLITE_ENABLE_DBSTAT_VTAB
    SQLITE_DEFAULT_WAL_SYNCHRONOUS=2 SQLITE_THREADSAFE=2)
install(TARGETS sqlite3 ARCHIVE DESTINATION lib)
install(FILES "${SQLITE_INCLUDE}/sqlite3.h" "${SQLITE_INCLUDE}/sqlite3ext.h" DESTINATION include)
